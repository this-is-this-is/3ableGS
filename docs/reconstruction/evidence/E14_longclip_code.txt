   187	def clip_image_embeddings(bundle: ClipBundle, paths: list[Path], device, batch_size: int):
   188	    """Raw projected CLIP image embeddings, in the order given.
   189	
   190	    Do not normalize each embedding here. CLIP-S applies cosine similarity
   191	    directly, while the paper's CLIP-DS formula takes the difference of the raw
   192	    embeddings before applying cosine similarity.
   193	    """
   194	    out = []
   195	    for start in range(0, len(paths), batch_size):
   196	        images = [Image.open(p).convert("RGB") for p in paths[start : start + batch_size]]
   197	        if bundle.backend == "longclip":
   198	            inputs = torch.stack([bundle.preprocess(image) for image in images]).to(device)
   199	            feats = bundle.model.encode_image(inputs)
   200	        else:
   201	            inputs = bundle.processor(images=images, return_tensors="pt").to(device)
   202	            feats = bundle.model.get_image_features(**inputs)
   203	            if hasattr(feats, "pooler_output"):
   204	                feats = feats.pooler_output
   205	        out.append(feats.cpu())
   206	    return torch.cat(out, dim=0)
