기존에는 segmentation 전처리기가 내부에서 다시 512×512로 바꾸는 문제가 있어 [eval_weather_metrics.py](/home/user/Desktop/WeatherCity/tools/eval_weather_metrics.py:173)를 수정했고, 기본 크기도 384×384로 변경했습니다. 실제 processor 입력이 `(1, 3, 384, 384)`인지 검증했습니다.

788씬 평가는 다음 명령어로 실행하면 됩니다.

```bash
cd /home/user/Desktop/WeatherCity
mkdir -p logs .cache/matplotlib

scene=788

SCENE="$scene" nohup bash \
  > "logs/scene${scene}_base_v4_weather_quality_paper_particle_free.log" 2>&1 <<'BASH' &

set -euo pipefail

export CUDA_VISIBLE_DEVICES=0
export PYTHONPATH="$PWD${PYTHONPATH:+:$PYTHONPATH}"
export MPLCONFIGDIR="$PWD/.cache/matplotlib"
export HF_HUB_OFFLINE=1
export TRANSFORMERS_OFFLINE=1

scene="${SCENE}"
run_root="output/pbr/base_v4/scene_${scene}_100f"
eval_root="${run_root}/weather_quality_paper_particle_free"

mkdir -p "${eval_root}"

for weather in rain snow
do
  if [ "${weather}" = "rain" ]; then
    ckpt="${run_root}/03_rain_joint/checkpoint_final.pth"
  else
    ckpt="${run_root}/06_snow_joint/checkpoint_final.pth"
  fi

  render_dir="${eval_root}/${weather}/frames"
  metric_file="${eval_root}/${weather}/metrics.json"

  mkdir -p "${render_dir}"

  # Weather material decoder + environment만 렌더링합니다.
  # particle renderer는 호출하지 않습니다.
  .conda/bin/python -u tools/export_rgb_sequence.py \
    --ckpt "${ckpt}" \
    --weather "${weather}" \
    --frames 0-29 \
    --out "${render_dir}"

  .conda/bin/python -u tools/eval_weather_metrics.py \
    --data_root data/waymo/processed/training \
    --scenes "${scene}" \
    --weathers "${weather}" \
    --cam 0 \
    --frames 30 \
    --raw_dirname raw \
    --edit_dir "${render_dir}" \
    --clip_model openai/clip-vit-large-patch14 \
    --seg_model openmmlab/upernet-convnext-xlarge \
    --prompt_mode full \
    --directional_mode paper \
    --seg_height 384 \
    --seg_width 384 \
    --batch_size 16 \
    --device cuda \
    --out "${metric_file}"
done
BASH

echo "PID=$!"
```
