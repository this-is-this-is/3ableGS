"""Resolve independent Base v3 boundary experiments; train only with --train."""
import argparse
from pathlib import Path
import subprocess
import sys
from omegaconf import OmegaConf

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
from tools.train import load_config_with_base
from models.boundary_losses import validate_boundary_config
from models.gaussians.boundary_renderer import validate_boundary_renderer_config

METHODS = (
    'baseline', 'antialias', 'boundary_rgb', 'boundary_perceptual',
    'boundary_perceptual_light', 'gbuffer_boundary', 'densify_late', 'densify_25k',
    'combo_lpips_25k', 'combo_depth_02', 'combo_boundary_grad_005',
    'combo_perceptual_02',
    'base_lr_decay', 'base_train_40k', 'base_lpips_crop_256',
    'base_lpips_late_decay', 'base_rgb075_ssim025',
    'base_hard_frame_sampling', 'base_stop_prune_25k',
    'densify_threshold', 'densify_grad_04', 'opacity_reset_20k',
    'densify_size_002', 'ssim_03', 'cull_alpha_003', 'image_pose',
    'rolling_shutter', 'layered_pbr', 'pixel_sort',
)


def experiment_config(scene, method):
    if method not in METHODS:
        raise ValueError(f'unknown method: {method}')
    base = load_config_with_base(str(ROOT / 'configs/base_v3' / f'scene{int(scene):03d}.yaml'))
    overlay = OmegaConf.load(ROOT / 'configs/base_v3_boundaries' / f'{method}.yaml')
    del overlay['base_config']
    cfg = OmegaConf.merge(base, overlay)
    validate_boundary_config(cfg.trainer.losses.get('boundary'))
    validate_boundary_config(cfg.trainer.losses.get('gbuffer_boundary'), gbuffer=True)
    validate_boundary_renderer_config(cfg.trainer.render.get('boundary_renderer'))
    return cfg


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--scene', type=int, choices=(23, 94, 114, 172, 788), default=788)
    parser.add_argument('--method', choices=METHODS, required=True)
    parser.add_argument('--output-root', default='output')
    parser.add_argument('--train', action='store_true')
    args = parser.parse_args()
    cfg = experiment_config(args.scene, args.method)
    if not args.train:
        print(OmegaConf.to_yaml(cfg))
        return
    # Fail before dataset loading if image-time metadata is missing.
    if args.method in ('image_pose', 'rolling_shutter'):
        from datasets.waymo.camera_motion import load_motion_record
        scene_dir = ROOT / cfg.data.data_root / f'{args.scene:03d}'
        for frame in range(cfg.data.start_timestep, cfg.data.end_timestep+1):
            for camera in cfg.data.pixel_source.cameras:
                load_motion_record(scene_dir / 'camera_motion' / f'{frame:03d}_{camera}.json', frame, camera)
    project = f'pbr/base_v3_boundaries/scene_{args.scene:03d}_100f'
    directory = ROOT / args.output_root / project / args.method
    if directory.exists() and any(directory.iterdir()):
        raise FileExistsError(f'{directory} is not empty; use a different --output-root to keep runs separate')
    directory.mkdir(parents=True, exist_ok=True)
    path = directory / 'experiment.yaml'
    OmegaConf.save(cfg, path)
    subprocess.run([sys.executable, '-u', 'tools/train.py', '--config_file', str(path),
                    '--output_root', args.output_root, '--project', project, '--run_name', args.method,
                    '--save_training_state'], cwd=ROOT, check=True)


if __name__ == '__main__':
    main()
