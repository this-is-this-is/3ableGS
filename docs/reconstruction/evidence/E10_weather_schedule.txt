확인 결과 rain/snow RGB 각 100장, G-buffer 각 400장, `combo_perceptual_02` 체크포인트 모두 준비되어 있습니다. 아래 명령은 해당 체크포인트를 Base v4 raw로 사용해 rain → snow 순서로 학습하며, 중단 후 같은 명령을 다시 실행하면 체크포인트부터 재개합니다.

```bash
cd /home/user/Desktop/WeatherCity
mkdir -p logs .cache/matplotlib

setsid nohup bash -lc '
set -euo pipefail
cd /home/user/Desktop/WeatherCity

export CUDA_VISIBLE_DEVICES=0
export MPLCONFIGDIR="$PWD/.cache/matplotlib"
export PYTHONPATH="$PWD"

CONFIG="configs/weathercity_pbr_snow.yaml"
RAW_CKPT="output/pbr/base_v3_boundaries/scene_788_100f/combo_perceptual_02/checkpoint_final.pth"
PROJECT="pbr/base_v4/scene_788_100f"

test -s "$RAW_CKPT"

run_stage() {
    local weather="$1"
    local stage="$2"
    local run_name="$3"
    local init_ckpt="$4"
    local iterations="$5"
    local save_freq="$6"
    shift 6

    local run_dir="output/$PROJECT/$run_name"
    local source_args=()

    if [ -s "$run_dir/checkpoint_final.pth" ]; then
        echo "[$(date -Is)] $run_name already completed"
        return
    fi

    if [ -s "$run_dir/checkpoint_training_latest.pth" ]; then
        source_args=(
            --resume_from "$run_dir/checkpoint_training_latest.pth"
            --resume_optimizer
        )
        echo "[$(date -Is)] resuming $run_name"
    else
        test -s "$init_ckpt"
        source_args=(--init_from_base "$init_ckpt")
        echo "[$(date -Is)] starting $run_name"
    fi

    .conda/bin/python -u tools/train.py \
        --config_file "$CONFIG" \
        --output_root output \
        --project "$PROJECT" \
        --run_name "$run_name" \
        --train_weathers "$weather" \
        --weather_stage "$stage" \
        --save_training_state \
        "${source_args[@]}" \
        trainer.optim.num_iters="$iterations" \
        logging.saveckpt_freq="$save_freq" \
        trainer.weather_training.weather="$weather" \
        model.EnvLight.params.weather_residuals="[$weather]" \
        data.pixel_source.load_weather="[$weather]" \
        data.pixel_source.gbuffer_path="data/waymo/render_png_788_single_static_100f/gbuffer_frames" \
        trainer.weather_training.paired_target_mode=gt_relative \
        trainer.weather_training.material_decoder.hidden_dim=64 \
        trainer.weather_training.material_decoder.num_hidden_layers=2 \
        data.preload_device=cpu \
        logging.vis_freq=0 \
        logging.print_freq=250 \
        render.render_test=false \
        render.render_full=false \
        render.render_novel=null \
        "$@"
}

for WEATHER in rain snow; do
    if [ "$WEATHER" = rain ]; then
        MAT_NAME="01_rain_material"
        ENV_NAME="02_rain_environment"
        JOINT_NAME="03_rain_joint"
    else
        MAT_NAME="04_snow_material"
        ENV_NAME="05_snow_environment"
        JOINT_NAME="06_snow_joint"
    fi

    run_stage "$WEATHER" material "$MAT_NAME" \
        "$RAW_CKPT" 30000 5000

    run_stage "$WEATHER" environment "$ENV_NAME" \
        "output/$PROJECT/$MAT_NAME/checkpoint_final.pth" 3000 1000 \
        trainer.weather_training.material_lr=0.0001 \
        trainer.weather_training.environment_lr=0.001 \
        trainer.res_schedule.double_steps=250 \
        trainer.res_schedule.downscale_times=2

    run_stage "$WEATHER" joint "$JOINT_NAME" \
        "output/$PROJECT/$ENV_NAME/checkpoint_final.pth" 5000 1000 \
        trainer.weather_training.material_lr=0.00005 \
        trainer.weather_training.environment_lr=0.0005 \
        trainer.res_schedule.double_steps=250 \
        trainer.res_schedule.downscale_times=2
done

echo "[$(date -Is)] scene 788 Base v4 rain/snow training completed"
' >> logs/scene788_base_v4_rain_snow.log 2>&1 < /dev/null &

PID=$!
echo "$PID" > logs/scene788_base_v4_rain_snow.pid
echo "PID: $PID"
```
